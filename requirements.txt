# Python packages. Installed by setup.bat — you should not need to run this
# by hand.
#
# Tested on Python 3.10 (Windows 10 / 11).

# --- Core -------------------------------------------------------------------
PyYAML                      # reads character_config.yaml
python-dotenv               # reads .env
requests                    # HTTP between the pieces

# --- The server the browser talks to ----------------------------------------
fastapi[standard]>=0.115.2
uvicorn

# --- Language models --------------------------------------------------------
openai                      # OpenAI-compatible internal tools and memory clients
groq                        # speech-to-text, and the tool-picking agent

# --- Voice out --------------------------------------------------------------
# --- Voice in ---------------------------------------------------------------
sounddevice                 # microphone capture
soundfile                   # reading and writing wav files
numpy
faster-whisper              # ONLY for the speech detector (Silero VAD) that
                            # decides when you stopped talking. Transcription
                            # itself goes to Groq. Set ENABLE_SILERO_VAD=False
                            # in server/config.py if you would rather not have
                            # this dependency — turn-taking gets a bit worse.

# --- Matching <expr:> and <emote:> tags to real animations ------------------
sentence-transformers       # the small model in models/all-MiniLM-L6-v2

# --- Memory (off by default — server/config.py) -----------------------------
faiss-cpu                   # stores and searches long-term memories

# --- Seeing your desktop ----------------------------------------------------
mss                         # screenshots
pillow                      # shrinks screenshots before the vision model
